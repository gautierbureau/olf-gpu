# Native build for OLF's GPU Newton-Raphson (same packaging pattern as
# powsybl-math-native): builds the JNI shared library libolfgpu into
# target/classes/natives/<arch>/, where it is picked up as a classpath resource by
# org.scijava.nativelib.NativeLoader (see
# com.powsybl.openloadflow.ac.equations.vector.gpu.GpuAcNewtonSolver) and bundled
# into the jar by the ordinary `mvn package`.
#
# This build is OPT-IN: the normal Maven build never invokes CMake, so the standard
# (no-CUDA) build and CI are completely unaffected — without this build there is
# simply no natives/ directory and the GPU code paths report isAvailable() == false.
# Building requires the CUDA toolkit AND cuDSS:
#
#   cmake -S . -B native/build -DCMAKE_BUILD_TYPE=Release \
#         -DCMAKE_CUDA_COMPILER=/path/to/cuda/bin/nvcc \
#         -DCMAKE_CUDA_ARCHITECTURES=86 \
#         -DCUDSS_ROOT=/path/to/cudss
#   cmake --build native/build
#
# (or just run native/build-gpu.sh, which defaults all of the above).
#
# DISTRIBUTION: libolfgpu is OUR wrapper (MPL) and ships in the jar; only NVIDIA's
# libcudss.so itself may not. cudart is linked STATICALLY; cuDSS is linked
# DYNAMICALLY (a NEEDED reference, never bundled — the same licensing posture as
# powsybl-math-native's libmathcudss). The default build is location-independent
# (no RPATH): the dynamic loader resolves libcudss/libcublas through LD_LIBRARY_PATH
# or the ldconfig cache when GpuNativeLoader loads libolfgpu from the classpath.
# CUDSS_EMBED_RPATH instead bakes an absolute DT_RPATH
# (--disable-new-dtags) to the cuDSS + CUDA toolkit lib dirs — only for non-JVM
# consumers of a LOCAL build, never for a jar meant to leave this machine.

cmake_minimum_required(VERSION 3.18.0)

if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 86)
endif()

project(olfgpu LANGUAGES C CXX CUDA)

find_package(JNI REQUIRED)
find_package(CUDAToolkit REQUIRED)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

set(CUDSS_ROOT "" CACHE PATH "Path to the cuDSS install (contains include/ and lib/)")
find_path(CUDSS_INCLUDE_DIR cudss.h HINTS ${CUDSS_ROOT}/include REQUIRED)
find_library(CUDSS_LIBRARY cudss HINTS ${CUDSS_ROOT}/lib REQUIRED)
message(STATUS "cuDSS header: ${CUDSS_INCLUDE_DIR}")
message(STATUS "cuDSS library: ${CUDSS_LIBRARY}")

# natives/<arch> layout expected by scijava native-lib-loader (same detection
# as powsybl-math-native's CMakeLists).
if(${CMAKE_SYSTEM_PROCESSOR} STREQUAL "x86_64" OR ${CMAKE_SYSTEM_PROCESSOR} STREQUAL "AMD64")
    set(OS_BITS 64)
elseif(${CMAKE_SYSTEM_PROCESSOR} STREQUAL "arm64")
    set(OS_BITS arm64)
else()
    set(OS_BITS 32)
endif()

if(${CMAKE_SYSTEM_NAME} STREQUAL "Windows")
    set(ARCHITECTURE windows_${OS_BITS})
elseif(${CMAKE_SYSTEM_NAME} STREQUAL "Linux")
    set(ARCHITECTURE linux_${OS_BITS})
elseif(${CMAKE_SYSTEM_NAME} STREQUAL "Darwin")
    set(ARCHITECTURE osx_${OS_BITS})
else()
    message(FATAL_ERROR "System not supported: ${CMAKE_SYSTEM_NAME}")
endif()

# native/cuda mirrors OLF's layout (common, diagnostics, equations, fixedmatrix, outerloop, sa,
# solver, structure), plus lu (our batched sparse LU); the JNI entry points live in native/cuda/jni/.
# The fused formula kernels (closed_branch.cuh, open_branch.cuh, shunt.cuh, hvdc_ac_emulation.cuh, load_model.cuh + their scatter tables)
# are GENERATED by the codegen (codegen/regenerate.sh) and committed in
# native/cuda/equations/generated/; every #include names them by a relative path.
# olf_gpu_violations.cu is a SEPARATE translation unit on purpose: it needs no cuDSS and nothing
# from the solver's context, only the generated flow formulas — so it compiles in seconds and
# cannot perturb the solver's codegen (adding a parameter to a solve kernel has flipped a 1e-12
# test before).
# SuiteSparse's approximate minimum degree (AMD 3.3.3 of SuiteSparse 7.8.3, BSD-3-clause, by Timothy A. Davis, Patrick R.
# Amestoy and Iain S. Duff): the elimination order our batched LU is planned on, computed without cuDSS. The release
# archive is downloaded at configure time, as powsybl-math-native does for KLU, and checked against its sha256; only
# the 32-bit integer AMD and SuiteSparse_config are compiled, unmodified, into libolfgpu with hidden visibility, so
# they never clash with another AMD loaded in the same JVM. Offline: -DFETCHCONTENT_SOURCE_DIR_SUITESPARSE=<unpacked
# archive>.
include(FetchContent)
set(SUITESPARSE_VERSION 7.8.3)
FetchContent_Declare(suitesparse
    URL https://github.com/DrTimothyAldenDavis/SuiteSparse/archive/refs/tags/v${SUITESPARSE_VERSION}.tar.gz
    URL_HASH SHA256=ce39b28d4038a09c14f21e02c664401be73c0cb96a9198418d6a98a7db73a259
)
FetchContent_GetProperties(suitesparse)
if(NOT suitesparse_POPULATED)
    FetchContent_Populate(suitesparse)   # sources only: SuiteSparse's own CMake build is not used
endif()
set(SUITESPARSE_AMD_SOURCES ${suitesparse_SOURCE_DIR}/SuiteSparse_config/SuiteSparse_config.c)
foreach(unit 1 2 aat control defaults info order postorder post_tree preprocess valid version)
    list(APPEND SUITESPARSE_AMD_SOURCES ${suitesparse_SOURCE_DIR}/AMD/Source/amd_${unit}.c)
endforeach()
set(SUITESPARSE_AMD_INCLUDE_DIRS ${suitesparse_SOURCE_DIR}/AMD/Include ${suitesparse_SOURCE_DIR}/SuiteSparse_config)

# cuBLAS supplies the BATCHED dense LU the SecondaryVoltageControl phase solves its per-scenario zone
# systems with (cublasDgetrfBatched/cublasDgetrsBatched) — many small matrices is exactly its case, and
# no linear algebra is written by hand. It is already present wherever cuDSS is (cuDSS links it).
target_link_libraries(olfgpu PRIVATE ${CUDSS_LIBRARY} CUDA::cublas)
# cuDSS >= 0.7.1 is required
# (static_assert in native/cuda/common/check.cuh); the cudssMatrixCreate* data-type args are shimmed by
# CUDSS_VERSION so the same source builds against 0.7.x and 0.8 (CUDSS_ROOT picks the install).
set_target_properties(olfgpu PROPERTIES CUDA_RUNTIME_LIBRARY Static)
# Only the JNI entry points (JNIEXPORT) are exported; everything in namespace olfgpu stays internal to the library.
set_target_properties(olfgpu PROPERTIES C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden CUDA_VISIBILITY_PRESET hidden
                      VISIBILITY_INLINES_HIDDEN ON)
# No implicit build-tree RUNPATH: the artifact is used straight from the build
# output (never `cmake --install`ed), and a baked machine-local path would defeat
# the distributable no-RPATH default. CUDSS_EMBED_RPATH adds it back explicitly.
set_target_properties(olfgpu PROPERTIES SKIP_BUILD_RPATH TRUE)

option(CUDSS_EMBED_RPATH "Embed an absolute RPATH to cuDSS/CUDA so libolfgpu runs without LD_LIBRARY_PATH (local builds only)" OFF)
if(CUDSS_EMBED_RPATH)
    get_filename_component(CUDSS_LIBRARY_DIR "${CUDSS_LIBRARY}" DIRECTORY)
    target_link_options(olfgpu PRIVATE
        "-Wl,--disable-new-dtags"
        "-Wl,-rpath,${CUDSS_LIBRARY_DIR}:${CUDAToolkit_LIBRARY_DIR}")
endif()

set(OLF_NATIVE_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/target/classes/natives/${ARCHITECTURE})
set_target_properties(olfgpu PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE ${OLF_NATIVE_OUTPUT_DIRECTORY})
set_target_properties(olfgpu PROPERTIES RUNTIME_OUTPUT_DIRECTORY_DEBUG ${OLF_NATIVE_OUTPUT_DIRECTORY})
set_target_properties(olfgpu PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${OLF_NATIVE_OUTPUT_DIRECTORY})
# AMD's license travels with the library that contains it (BSD-3-clause: binary redistributions carry the notice).
